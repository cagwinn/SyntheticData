```rust
/// Breakdown of production order costs into constituent components.
///
/// Enables variance analysis: actual vs standard for each component.
#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct CostBreakdown {
    /// Material cost (raw materials consumed)
    #[serde(with = "rust_decimal::serde::str")]
    pub material_cost: Decimal,
    /// Direct labor cost (hours × rate)
    #[serde(with = "rust_decimal::serde::str")]
    pub labor_cost: Decimal,
    /// Applied overhead (labor cost × overhead rate)
    #[serde(with = "rust_decimal::serde::str")]
    pub overhead_cost: Decimal,
    /// Standard material cost for variance calculation
    #[serde(with = "rust_decimal::serde::str")]
    pub standard_material_cost: Decimal,
    /// Standard labor cost for variance calculation
    #[serde(with = "rust_decimal::serde::str")]
    pub standard_labor_cost: Decimal,
    /// Standard overhead cost for variance calculation
    #[serde(with = "rust_decimal::serde::str")]
    pub standard_overhead_cost: Decimal,
    /// Standard cost per unit of output
    #[serde(with = "rust_decimal::serde::str")]
    pub standard_unit_cost: Decimal,
}

impl CostBreakdown {
    /// Total actual cost (sum of material + labor + overhead).
    pub fn total_actual(&self) -> Decimal {
        self.material_cost + self.labor_cost + self.overhead_cost
    }

    /// Total standard cost.
    pub fn total_standard(&self) -> Decimal {
        self.standard_material_cost + self.standard_labor_cost + self.standard_overhead_cost
    }

    /// Total variance (actual - standard).
    pub fn total_variance(&self) -> Decimal {
        self.total_actual() - self.total_standard()
    }
}
```

```rust
//! Multi-stage manufacturing cost accounting JE pipeline.
//!
//! Replaces the flat "DR Raw Materials / CR Inventory" with proper cost flow:
//! 1. Order started   → DR WIP, CR Raw Materials (material issue)
//! 2. Operations done  → DR WIP, CR Labor Accrual (labor absorption)
//! 3. Order completed  → DR Finished Goods, CR WIP (FG transfer)
//! 4. Inspection failed → DR Scrap Expense, CR WIP or FG
//! 5. Order settled    → Variance JEs (material price/usage, labor rate/efficiency, overhead volume)

use chrono::NaiveDate;
use datasynth_core::accounts::{control_accounts, manufacturing_accounts};
use datasynth_core::models::{
    CostBreakdown, JournalEntry, JournalEntryLine, ProductionOrder,
    ProductionOrderStatus, QualityInspection, InspectionResult,
};
use rust_decimal::Decimal;

/// Generates the full set of manufacturing cost JEs for a generation run.
pub struct ManufacturingCostAccounting;

impl ManufacturingCostAccounting {
    /// Generate all manufacturing JEs from production orders and quality inspections.
    ///
    /// Returns JEs in lifecycle order: material issue → labor → FG transfer → scrap → variances.
    pub fn generate_all_jes(
        orders: &[ProductionOrder],
        inspections: &[QualityInspection],
        _currency: &str,
    ) -> Vec<JournalEntry> {
        let mut jes = Vec::new();

        for order in orders {
            // Skip planned/cancelled orders — no cost flow
            if matches!(order.status, ProductionOrderStatus::Planned | ProductionOrderStatus::Cancelled) {
                continue;
            }

            let breakdown = match &order.cost_breakdown {
                Some(b) => b,
                None => continue, // legacy orders without breakdown — skip
            };

            let posting_date = order.actual_end.unwrap_or(order.planned_end);

            // Stage 1: Material issue to WIP (Released, InProcess, Completed, Closed)
            if breakdown.material_cost > Decimal::ZERO {
                jes.push(Self::material_issue_je(order, breakdown, posting_date));
            }

            // Stage 2: Labor absorption to WIP
            if breakdown.labor_cost > Decimal::ZERO {
                jes.push(Self::labor_absorption_je(order, breakdown, posting_date));
            }

            // Stage 3: Overhead applied to WIP
            if breakdown.overhead_cost > Decimal::ZERO {
                jes.push(Self::overhead_applied_je(order, breakdown, posting_date));
            }

            // Stage 4: FG transfer (only Completed/Closed)
            if matches!(order.status, ProductionOrderStatus::Completed | ProductionOrderStatus::Closed) {
                jes.push(Self::fg_transfer_je(order, breakdown, posting_date));
            }

            // Stage 5: Scrap from quality inspection
            let order_inspections: Vec<_> = inspections.iter()
                .filter(|i| i.reference_id == order.order_id)
                .collect();
            for insp in &order_inspections {
                if matches!(insp.result, InspectionResult::Rejected) {
                    if let Some(scrap_je) = Self::scrap_je(order, insp, posting_date) {
                        jes.push(scrap_je);
                    }
                }
            }

            // Stage 6: Variance JEs (only Completed/Closed — settlement)
            if matches!(order.status, ProductionOrderStatus::Completed | ProductionOrderStatus::Closed) {
                jes.extend(Self::variance_jes(order, breakdown, posting_date));
            }
        }

        jes
    }

    fn material_issue_je(order: &ProductionOrder, bd: &CostBreakdown, date: NaiveDate) -> JournalEntry {
        let mut je = JournalEntry::new_simple(
            format!("JE-MFG-MAT-{}", order.order_id),
            order.company_code.clone(),
            date,
            format!("WIP material issue — {}", order.material_description),
        );
        je.add_line(JournalEntryLine {
            line_number: 1,
            gl_account: manufacturing_accounts::WIP.to_string(),
            debit_amount: bd.material_cost,
            reference: Some(order.order_id.clone()),
            text: Some(format!("Material issue to WIP for {}", order.order_id)),
            quantity: Some(order.planned_quantity),
            unit: Some("EA".to_string()),
            ..Default::default()
        });
        je.add_line(JournalEntryLine {
            line_number: 2,
            gl_account: control_accounts::INVENTORY.to_string(),
            credit_amount: bd.material_cost,
            reference: Some(order.order_id.clone()),
            ..Default::default()
        });
        je
    }

    fn labor_absorption_je(order: &ProductionOrder, bd: &CostBreakdown, date: NaiveDate) -> JournalEntry {
        let mut je = JournalEntry::new_simple(
            format!("JE-MFG-LAB-{}", order.order_id),
            order.company_code.clone(),
            date,
            format!("WIP labor absorption — {}", order.material_description),
        );
        je.add_line(JournalEntryLine {
            line_number: 1,
            gl_account: manufacturing_accounts::WIP.to_string(),
            debit_amount: bd.labor_cost,
            reference: Some(order.order_id.clone()),
            text: Some(format!("Labor absorbed: {:.1}h", order.labor_hours)),
            ..Default::default()
        });
        je.add_line(JournalEntryLine {
            line_number: 2,

    fn scrap_je(order: &ProductionOrder, inspection: &QualityInspection, date: NaiveDate) -> Option<JournalEntry> {
        if order.scrap_quantity <= Decimal::ZERO {
            return None;
        }
        let bd = order.cost_breakdown.as_ref()?;
        // Scrap value = scrap_quantity / planned_quantity * total_actual
        let scrap_fraction = if order.planned_quantity > Decimal::ZERO {
            order.scrap_quantity / order.planned_quantity
        } else {
            Decimal::ZERO
        };
        let scrap_value = (bd.total_actual() * scrap_fraction).round_dp(2);
        if scrap_value <= Decimal::ZERO {
            return None;
        }

        let mut je = JournalEntry::new_simple(
            format!("JE-MFG-SCRAP-{}", inspection.inspection_id),
            order.company_code.clone(),
            date,
            format!("Scrap write-off — {} (QI: {})", order.material_description, inspection.inspection_id),
        );
        je.add_line(JournalEntryLine {
            line_number: 1,
            gl_account: manufacturing_accounts::SCRAP_EXPENSE.to_string(),
            debit_amount: scrap_value,
            reference: Some(order.order_id.clone()),
            quantity: Some(order.scrap_quantity),
            unit: Some("EA".to_string()),
            ..Default::default()
        });
        // Credit WIP if order still in process, FG if completed
        let credit_account = if matches!(order.status, ProductionOrderStatus::Completed | ProductionOrderStatus::Closed) {
            manufacturing_accounts::FINISHED_GOODS
        } else {
            manufacturing_accounts::WIP
        };
        je.add_line(JournalEntryLine {
            line_number: 2,
            gl_account: credit_account.to_string(),
            credit_amount: scrap_value,
            reference: Some(order.order_id.clone()),
            ..Default::default()
        });
        Some(je)
    }

    fn variance_jes(order: &ProductionOrder, bd: &CostBreakdown, date: NaiveDate) -> Vec<JournalEntry> {
        let mut jes = Vec::new();
        let variance_entries = [
            ("material price", bd.material_cost - bd.standard_material_cost, manufacturing_accounts::MATERIAL_PRICE_VARIANCE),
            ("labor rate", bd.labor_cost - bd.standard_labor_cost, manufacturing_accounts::LABOR_RATE_VARIANCE),
            ("overhead volume", bd.overhead_cost - bd.standard_overhead_cost, manufacturing_accounts::OVERHEAD_VOLUME_VARIANCE),
        ];

        for (label, variance, account) in &variance_entries {
            let variance = variance.round_dp(2);
            if variance == Decimal::ZERO {
                continue;
            }
            let mut je = JournalEntry::new_simple(
                format!("JE-MFG-VAR-{}-{}", label.replace(' ', "-"), order.order_id),
                order.company_code.clone(),
                date,
                format!("Mfg {} variance — {}", label, order.order_id),
            );
            if variance > Decimal::ZERO {
                // Unfavorable: actual > standard → debit variance, credit WIP
                je.add_line(JournalEntryLine {
                    line_number: 1,
                    gl_account: account.to_string(),
                    debit_amount: variance,
                    reference: Some(order.order_id.clone()),
                    ..Default::default()
                });
                je.add_line(JournalEntryLine {
                    line_number: 2,
                    gl_account: manufacturing_accounts::WIP.to_string(),
                    credit_amount: variance,
                    reference: Some(order.order_id.clone()),
                    ..Default::default()
                });
            } else {
                // Favorable: actual < standard → credit variance, debit WIP
                let abs_variance = variance.abs();
                je.add_line(JournalEntryLine {
                    line_number: 1,
                    gl_account: manufacturing_accounts::WIP.to_string(),
                    debit_amount: abs_variance,
                    reference: Some(order.order_id.clone()),
                    ..Default::default()
                });
                je.add_line(JournalEntryLine {
                    line_number: 2,
                    gl_account: account.to_string(),
                    credit_amount: abs_variance,
                    reference: Some(order.order_id.clone()),
                    ..Default::default()
                });
            }
            jes.push(je);
        }
        jes
    }
}
```

```rust
//! Warranty provision generator — creates provisions from quality failure patterns.
//!
//! Links: QualityInspection failures → Provision (IAS 37 / ASC 450) → ProvisionMovement → JEs.

use chrono::NaiveDate;
use datasynth_core::accounts::manufacturing_accounts;
use datasynth_core::models::{
    JournalEntry, JournalEntryLine, ProductionOrder, ProductionOrderStatus,
    QualityInspection, InspectionResult,
};
use datasynth_core::models::provision::{Provision, ProvisionMovement, ProvisionType};
use datasynth_core::uuid_factory::DeterministicUuidFactory;
use rand::Rng;
use rand_chacha::ChaCha8Rng;
use rand::SeedableRng;
use rust_decimal::Decimal;
use rust_decimal::prelude::FromPrimitive;

/// Output of warranty provision generation.
pub struct WarrantyProvisionResult {
    pub provisions: Vec<Provision>,
    pub movements: Vec<ProvisionMovement>,
    pub journal_entries: Vec<JournalEntry>,
}

/// Generates warranty provisions based on quality inspection failure rates.
pub struct WarrantyProvisionGenerator {
    rng: ChaCha8Rng,
    uuid_factory: DeterministicUuidFactory,
}

impl WarrantyProvisionGenerator {
    pub fn new(seed: u64) -> Self {
        Self {
            rng: ChaCha8Rng::seed_from_u64(seed),
            uuid_factory: DeterministicUuidFactory::new(seed, "WP"),
        }
    }

    /// Generate warranty provisions from completed production orders and their inspections.
    ///
    /// Failure rate from inspections drives the provision estimate:
    /// `provision = total_production_value × defect_rate × warranty_cost_factor`
    pub fn generate(
        &mut self,
        company_code: &str,
        orders: &[ProductionOrder],
        inspections: &[QualityInspection],
        currency: &str,
        framework: &str,
    ) -> WarrantyProvisionResult {
        let mut provisions = Vec::new();
        let mut movements = Vec::new();
        let mut journal_entries = Vec::new();

        // Calculate overall defect rate from inspections
        let total_inspected: usize = inspections.len();
        let total_rejected: usize = inspections.iter()
            .filter(|i| matches!(i.result, InspectionResult::Rejected))
            .count();
        if total_inspected == 0 {
            return WarrantyProvisionResult { provisions, movements, journal_entries };
        }
        let defect_rate = total_rejected as f64 / total_inspected as f64;

        // Only create provision if defect rate is material (> 1%)
        if defect_rate < 0.01 {
            return WarrantyProvisionResult { provisions, movements, journal_entries };
        }

        // Total production value from completed orders
        let completed_orders: Vec<_> = orders.iter()
            .filter(|o| matches!(o.status, ProductionOrderStatus::Completed | ProductionOrderStatus::Closed))
            .collect();
        let total_production_value: Decimal = completed_orders.iter()
            .map(|o| o.actual_cost)
            .sum();

        if total_production_value <= Decimal::ZERO {
            return WarrantyProvisionResult { provisions, movements, journal_entries };
        }

        // Warranty cost factor: 1.2x-2.0x of defective unit cost (repair/replace + handling)
        let warranty_cost_factor: f64 = self.rng.random_range(1.2..=2.0);
        let defect_rate_dec = Decimal::from_f64_retain(defect_rate).unwrap_or(Decimal::ZERO);
        let factor_dec = Decimal::from_f64_retain(warranty_cost_factor).unwrap_or(Decimal::ONE);

        let best_estimate = (total_production_value * defect_rate_dec * factor_dec).round_dp(2);
        let range_low = (best_estimate * Decimal::from_f64_retain(0.7).unwrap()).round_dp(2);
        let range_high = (best_estimate * Decimal::from_f64_retain(1.5).unwrap()).round_dp(2);

        // Determine posting date from latest completed order
        let posting_date = completed_orders.iter()
            .filter_map(|o| o.actual_end)
            .max()
            .unwrap_or_else(|| completed_orders.last().map(|o| o.planned_end)
                .unwrap_or(NaiveDate::from_ymd_opt(2025, 12, 31).unwrap()));

        let provision_id = self.uuid_factory.create_v7();

        let provision = Provision {
            id: provision_id.clone(),
            entity_code: company_code.to_string(),
            provision_type: ProvisionType::Warranty,
            description: format!(
                "Product warranty — defect rate {:.1}% on production value {}",
                defect_rate * 100.0, total_production_value
            ),
            best_estimate,
            range_low,
            range_high,
            discount_rate: None, // Short-term, no discounting
            expected_utilization_date: posting_date + chrono::Duration::days(365),
            framework: framework.to_string(),
            currency: currency.to_string(),
        };

        // Provision movement (initial recognition)
        let period = format!("{}", posting_date.format("%Y-Q%q"));
        let movement = ProvisionMovement {
            provision_id: provision_id.clone(),
            period: period.clone(),
            opening: Decimal::ZERO,
            additions: best_estimate,
            utilizations: Decimal::ZERO,
            reversals: Decimal::ZERO,
            unwinding_of_discount: Decimal::ZERO,
            closing: best_estimate,
        };

        // Journal entry: DR Warranty Expense, CR Warranty Provision
        let mut je = JournalEntry::new_simple(
            format!("JE-WARRANTY-{}", provision_id),
            company_code.to_string(),
            posting_date,
            format!("Warranty provision recognition — defect rate {:.1}%", defect_rate * 100.0),
        );
        je.add_line(JournalEntryLine {
            line_number: 1,
            gl_account: manufacturing_accounts::WARRANTY_EXPENSE.to_string(),
            debit_amount: best_estimate,
            reference: Some(provision_id.clone()),
            text: Some("Warranty expense provision".to_string()),
            ..Default::default()
        });
        je.add_line(JournalEntryLine {
            line_number: 2,
            gl_account: manufacturing_accounts::WARRANTY_PROVISION.to_string(),
            credit_amount: best_estimate,
            reference: Some(provision_id.clone()),
            ..Default::default()
        });

        provisions.push(provision);
        movements.push(movement);
        journal_entries.push(je);

        WarrantyProvisionResult { provisions, movements, journal_entries }
    }
}
```

```rust
pub struct ICDocumentChains {

    ///
    /// For each ICMatchedPair of type GoodsSale or ServiceProvided:
    /// - Seller side: CustomerInvoice with `invoice_type: Intercompany`
    /// - Buyer side: PurchaseOrder → GoodsReceipt → VendorInvoice
    ///
    /// Each document's `reference` field contains the `ic_reference` for traceability.
    pub fn generate_ic_document_chains(&mut self, pairs: &[ICMatchedPair]) -> ICDocumentChains {
        let mut chains = ICDocumentChains {
            seller_invoices: Vec::new(),
            buyer_orders: Vec::new(),
            buyer_goods_receipts: Vec::new(),
            buyer_invoices: Vec::new(),
        };

        for pair in pairs {
            // Only generate docs for goods/service transactions (not loans, dividends, etc.)
            if !matches!(pair.transaction_type,
                ICTransactionType::GoodsSale | ICTransactionType::ServiceProvided |
                ICTransactionType::ManagementFee | ICTransactionType::Royalty |
                ICTransactionType::ExpenseRecharge
            ) {
                continue;
            }

            self.doc_counter += 1;
            let doc_num = self.doc_counter;

            // Seller-side: CustomerInvoice (IC type)
            let seller_invoice = self.build_ic_customer_invoice(pair, doc_num);

            let (po, gr, vi) = self.build_ic_purchase_chain(pair, doc_num);
            chains.buyer_orders.push(po);
            chains.buyer_goods_receipts.push(gr);
            chains.buyer_invoices.push(vi);
        }

        chains
    }
```

```rust
//! Inventory and COGS coherence validators.
//!
//! Validates:
//! - FG reconciliation: Opening FG + Completions - COGS - Scrap = Closing FG
//! - WIP reconciliation: Opening WIP + Issues + Labor + OH - Completions - Scrap = Closing WIP
//! - Variance reconciliation: Total variance = Sum of component variances

use rust_decimal::Decimal;
use serde::{Deserialize, Serialize};

/// Input data for inventory/COGS coherence validation.
#[derive(Debug, Clone)]
pub struct InventoryCOGSData {
    // Finished Goods reconciliation
    pub opening_fg: Decimal,
    pub production_completions: Decimal,
    pub cogs: Decimal,
    pub scrap: Decimal,
    pub closing_fg: Decimal,

    // WIP reconciliation
    pub opening_wip: Decimal,
    pub material_issues: Decimal,
    pub labor_absorbed: Decimal,
    pub overhead_applied: Decimal,
    pub completions_out_of_wip: Decimal,
    pub wip_scrap: Decimal,
    pub closing_wip: Decimal,

    // Variance reconciliation
    pub total_variance: Decimal,
    pub sum_of_component_variances: Decimal,
}

/// Results of inventory/COGS coherence evaluation.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct InventoryCOGSEvaluation {
    pub fg_reconciled: bool,
    pub fg_imbalance: Decimal,
    pub fg_expected_closing: Decimal,
    pub wip_reconciled: bool,
    pub wip_imbalance: Decimal,
    pub wip_expected_closing: Decimal,
    pub variances_reconciled: bool,
    pub variance_difference: Decimal,
    pub passes: bool,
    pub failures: Vec<String>,
}

/// Evaluator for inventory and COGS coherence.
pub struct InventoryCOGSEvaluator {
    tolerance: Decimal,
}

impl InventoryCOGSEvaluator {
    pub fn new(tolerance: Decimal) -> Self {
        Self { tolerance }
    }

    pub fn evaluate(&self, data: &InventoryCOGSData) -> InventoryCOGSEvaluation {
        let mut failures = Vec::new();

        // FG: Opening + Completions - COGS - Scrap = Closing
        let fg_expected = data.opening_fg + data.production_completions - data.cogs - data.scrap;
        let fg_imbalance = data.closing_fg - fg_expected;
        let fg_reconciled = fg_imbalance.abs() <= self.tolerance;
        if !fg_reconciled {
            failures.push(format!(
                "FG imbalance: expected closing {} but got {} (diff {})",
                fg_expected, data.closing_fg, fg_imbalance
            ));
        }

        // WIP: Opening + Material + Labor + OH - Completions - Scrap = Closing
        let wip_expected = data.opening_wip + data.material_issues + data.labor_absorbed
            + data.overhead_applied - data.completions_out_of_wip - data.wip_scrap;
        let wip_imbalance = data.closing_wip - wip_expected;
        let wip_reconciled = wip_imbalance.abs() <= self.tolerance;
        if !wip_reconciled {
            failures.push(format!(
                "WIP imbalance: expected closing {} but got {} (diff {})",
                wip_expected, data.closing_wip, wip_imbalance
            ));
        }

        // Variances: total = sum of components
        let variance_difference = data.total_variance - data.sum_of_component_variances;
        let variances_reconciled = variance_difference.abs() <= self.tolerance;
        if !variances_reconciled {
            failures.push(format!(
                "Variance mismatch: total {} vs components {} (diff {})",
                data.total_variance, data.sum_of_component_variances, variance_difference
            ));
        }

        let passes = failures.is_empty();

        InventoryCOGSEvaluation {
            fg_reconciled,
            fg_imbalance,
            fg_expected_closing: fg_expected,
            wip_reconciled,
            wip_imbalance,
            wip_expected_closing: wip_expected,
            variances_reconciled,
            variance_difference,
            passes,
            failures,
        }
    }
}
```

```rust
impl ManufacturingCostAccounting {
    /// Generate COGS JEs when goods are delivered/sold.
    ///
    /// For each Delivery, finds the matching FG cost (from completed production orders
    /// for the same material) and generates: DR COGS, CR Finished Goods.
    pub fn generate_cogs_on_sale(
        deliveries: &[Delivery],
        production_orders: &[ProductionOrder],
    ) -> Vec<JournalEntry> {
        use datasynth_core::accounts::expense_accounts;

        let mut jes = Vec::new();

        // Build material → average unit cost from completed orders
        let mut material_costs: std::collections::HashMap<String, (Decimal, Decimal)> = std::collections::HashMap::new();
        for order in production_orders {
            if !matches!(order.status, ProductionOrderStatus::Completed | ProductionOrderStatus::Closed) {
                continue;
            }
            if order.actual_quantity > Decimal::ZERO {
                let unit_cost = order.actual_cost / order.actual_quantity;
                let entry = material_costs.entry(order.material_id.clone()).or_insert((Decimal::ZERO, Decimal::ZERO));
                entry.0 += order.actual_cost;
                entry.1 += order.actual_quantity;
            }
        }

        for delivery in deliveries {
            for item in &delivery.items {
                let material_id = item.material_id.as_deref().unwrap_or("");
                if let Some((total_cost, total_qty)) = material_costs.get(material_id) {
                    if *total_qty > Decimal::ZERO {
                        let avg_unit_cost = (*total_cost / *total_qty).round_dp(2);
                        let cogs_amount = (avg_unit_cost * item.quantity).round_dp(2);

                        let mut je = JournalEntry::new_simple(
                            format!("JE-COGS-{}-{}", delivery.header.document_id, item.line_number),
                            delivery.header.company_code.clone(),
                            delivery.header.posting_date,
                            format!("COGS on delivery {} — {}", delivery.header.document_id, material_id),
                        );
                        je.add_line(JournalEntryLine {
                            line_number: 1,
                            gl_account: expense_accounts::COGS.to_string(),
                            debit_amount: cogs_amount,
                            reference: Some(delivery.header.document_id.clone()),
                            quantity: Some(item.quantity),
                            ..Default::default()
                        });
                        je.add_line(JournalEntryLine {
                            line_number: 2,
                            gl_account: manufacturing_accounts::FINISHED_GOODS.to_string(),
                            credit_amount: cogs_amount,
                            reference: Some(delivery.header.document_id.clone()),
                            ..Default::default()
                        });
                        jes.push(je);
                    }
                }
            }
        }
        jes
    }
}
```

```rust
/// Input data for IC elimination completeness validation.
#[derive(Debug, Clone)]
pub struct ICEliminationData {
    /// Number of IC matched pairs generated.
    pub matched_pair_count: usize,
    /// Number of elimination entries generated.
    pub elimination_entry_count: usize,
    /// Total IC transaction amount (sum of all matched pair amounts).
    pub total_ic_amount: Decimal,
    /// Total elimination amount (sum of all elimination entry debits).
    pub total_elimination_amount: Decimal,
    /// Number of IC matched pairs with corresponding elimination entries.
    pub pairs_with_eliminations: usize,
}

/// Results of IC elimination completeness evaluation.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ICEliminationEvaluation {
    pub all_pairs_eliminated: bool,
    pub coverage_rate: f64,
    pub amount_reconciled: bool,
    pub amount_difference: Decimal,
    pub passes: bool,
    pub failures: Vec<String>,
}

/// Evaluator for IC elimination completeness.
pub struct ICEliminationEvaluator {
    tolerance: Decimal,
}

impl ICEliminationEvaluator {
    pub fn new(tolerance: Decimal) -> Self {
        Self { tolerance }
    }

    pub fn evaluate(&self, data: &ICEliminationData) -> ICEliminationEvaluation {
        let mut failures = Vec::new();

        let coverage_rate = if data.matched_pair_count > 0 {
            data.pairs_with_eliminations as f64 / data.matched_pair_count as f64
        } else {
            1.0
        };

        let all_pairs_eliminated = data.pairs_with_eliminations >= data.matched_pair_count;
        if !all_pairs_eliminated {
            failures.push(format!(
                "IC coverage: only {}/{} pairs have eliminations ({:.1}%)",
                data.pairs_with_eliminations, data.matched_pair_count, coverage_rate * 100.0
            ));
        }

        let amount_difference = (data.total_ic_amount - data.total_elimination_amount).abs();
        let amount_reconciled = amount_difference <= self.tolerance;
        if !amount_reconciled {
            failures.push(format!(
                "IC amount mismatch: IC total {} vs elimination total {} (diff {})",
                data.total_ic_amount, data.total_elimination_amount, amount_difference
            ));
        }

        let passes = failures.is_empty();

        ICEliminationEvaluation {
            all_pairs_eliminated,
            coverage_rate,
            amount_reconciled,
            amount_difference,
            passes,
            failures,
        }
    }
}
```
