# v2.3 Treasury, Tax & Payroll JE Pipeline — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire treasury instruments, tax provisions, and payroll into the GL — every debt instrument accrues interest, every hedge has P&L impact, tax provisions derive from actual pre-tax income, and payroll reflects HR change events.

**Architecture:** Three independent workstreams (treasury JEs, tax linkage, payroll↔HR) share new GL account constants and converge in the orchestrator. Treasury creates a new `treasury_accounting.rs` module parallel to v2.2's `cost_accounting.rs`. Tax fixes the hardcoded pre-tax income and adds VAT posting JEs. Payroll modifies the existing generator to consume `EmployeeChangeHistory`.

**Tech Stack:** Rust, rust_decimal, chrono, serde, ChaCha8Rng

```rust
//! Treasury accounting JE pipeline.
//!
//! Generates journal entries from treasury instruments:
//! - Debt: interest accrual, amortization (effective interest), maturity
//! - Hedging: mark-to-market, OCI (cash flow hedge), P&L (fair value hedge), ineffectiveness
//! - Cash pools: physical pooling IC entries

use chrono::NaiveDate;
use datasynth_core::accounts::{cash_accounts, expense_accounts, liability_accounts, treasury_accounts};
use datasynth_core::models::treasury::{
    CashPoolSweep, DebtInstrument, HedgeRelationship, HedgeType, HedgingInstrument,
};
use datasynth_core::models::{JournalEntry, JournalEntryLine};
use rust_decimal::Decimal;
use rust_decimal::prelude::FromPrimitive;

pub struct TreasuryAccounting;

impl TreasuryAccounting {
    /// Generate all debt-related JEs for a period.
    ///
    /// For each debt instrument:
    /// - Interest accrual: DR Interest Expense, CR Interest Payable
    ///   Amount = principal × annual_rate × (days_in_period / 360)
    pub fn generate_debt_jes(
        instruments: &[DebtInstrument],
        period_end: NaiveDate,
    ) -> Vec<JournalEntry> {
        let mut jes = Vec::new();

        for debt in instruments {
            // Skip matured instruments
            if period_end > debt.maturity_date {
                continue;
            }

            // Interest accrual (quarterly approximation)
            let quarterly_rate = debt.interest_rate / Decimal::from(4);
            let interest_amount = (debt.principal * quarterly_rate).round_dp(2);

            if interest_amount > Decimal::ZERO {
                let mut je = JournalEntry::new_simple(
                    format!("JE-TREAS-INT-{}", debt.id),
                    debt.entity_id.clone(),
                    period_end,
                    format!("Interest accrual — {} ({})", debt.id, debt.lender),
                );
                je.add_line(JournalEntryLine {
                    line_number: 1,
                    gl_account: expense_accounts::INTEREST_EXPENSE.to_string(),
                    debit_amount: interest_amount,
                    reference: Some(debt.id.clone()),
                    text: Some(format!(
                        "Interest: {} × {:.4}% quarterly",
                        debt.principal, quarterly_rate
                    )),
                    ..Default::default()
                });
                je.add_line(JournalEntryLine {
                    line_number: 2,
                    gl_account: treasury_accounts::INTEREST_PAYABLE.to_string(),
                    credit_amount: interest_amount,
                    reference: Some(debt.id.clone()),
                    ..Default::default()
                });
                jes.push(je);
            }
        }

        jes
    }

    /// Generate hedge mark-to-market JEs.
    ///
    /// - Cash flow hedge: DR/CR Derivative Asset/Liability, CR/DR OCI
    /// - Fair value hedge: DR/CR Derivative Asset/Liability, CR/DR P&L
    /// - Ineffectiveness: excess goes to P&L
    pub fn generate_hedge_jes(
        instruments: &[HedgingInstrument],
        relationships: &[HedgeRelationship],
        period_end: NaiveDate,
    ) -> Vec<JournalEntry> {
        let mut jes = Vec::new();

        for instrument in instruments {
            if !instrument.is_active() {
                continue;
            }

            let fair_value = instrument.fair_value;
            if fair_value == Decimal::ZERO {
                continue;
            }

            // Find the hedge relationship for this instrument
            let relationship = relationships.iter()
                .find(|r| r.hedging_instrument_id == instrument.id);

            let hedge_type = relationship.map(|r| r.hedge_type).unwrap_or(HedgeType::FairValueHedge);
            let is_effective = relationship.map(|r| r.is_effective).unwrap_or(true);
            let ineffectiveness = relationship.map(|r| r.ineffectiveness_amount).unwrap_or(Decimal::ZERO);

            let abs_fv = fair_value.abs();

            // Derivative fair value JE
            let (derivative_account, offset_account) = if fair_value > Decimal::ZERO {
                // Positive FV → derivative is an asset
                match hedge_type {
                    HedgeType::CashFlowHedge => (
                        treasury_accounts::DERIVATIVE_ASSET,
                        treasury_accounts::OCI_CASH_FLOW_HEDGE,
                    ),
                    _ => (
                        treasury_accounts::DERIVATIVE_ASSET,
                        expense_accounts::FX_GAIN_LOSS, // P&L for fair value hedges
                    ),
                }
            } else {
                // Negative FV → derivative is a liability
                match hedge_type {
                    HedgeType::CashFlowHedge => (
                        treasury_accounts::OCI_CASH_FLOW_HEDGE,
                        treasury_accounts::DERIVATIVE_LIABILITY,
                    ),
                    _ => (
                        expense_accounts::FX_GAIN_LOSS,
                        treasury_accounts::DERIVATIVE_LIABILITY,
                    ),
                }
            };

            let mut je = JournalEntry::new_simple(
                format!("JE-TREAS-HEDGE-{}", instrument.id),
                "GROUP".to_string(), // Hedge JEs are at group level
                period_end,
                format!(
                    "Hedge MTM — {} ({:?})",
                    instrument.id,
                    hedge_type,
                ),
            );

            je.add_line(JournalEntryLine {
                line_number: 1,
                gl_account: derivative_account.to_string(),
                debit_amount: abs_fv,
                reference: Some(instrument.id.clone()),
                ..Default::default()
            });
            je.add_line(JournalEntryLine {
                line_number: 2,
                gl_account: offset_account.to_string(),
                credit_amount: abs_fv,
                reference: Some(instrument.id.clone()),
                ..Default::default()
            });
            jes.push(je);

            // Ineffectiveness JE (if not effective and amount is material)
            if !is_effective && ineffectiveness.abs() > Decimal::ZERO {
                let ineff = ineffectiveness.abs().round_dp(2);
                let mut ineff_je = JournalEntry::new_simple(
                    format!("JE-TREAS-INEFF-{}", instrument.id),
                    "GROUP".to_string(),
                    period_end,
                    format!("Hedge ineffectiveness — {}", instrument.id),
                );
                ineff_je.add_line(JournalEntryLine {
                    line_number: 1,
                    gl_account: treasury_accounts::HEDGE_INEFFECTIVENESS.to_string(),
                    debit_amount: ineff,
                    reference: Some(instrument.id.clone()),
                    ..Default::default()
                });
                ineff_je.add_line(JournalEntryLine {
                    line_number: 2,
                    gl_account: treasury_accounts::OCI_CASH_FLOW_HEDGE.to_string(),
                    credit_amount: ineff,
                    reference: Some(instrument.id.clone()),
                    ..Default::default()
                });
                jes.push(ineff_je);
            }
        }

        jes
    }

    /// Generate JEs for physical cash pool sweeps.
    ///
    /// Each sweep: DR/CR Cash Pool IC Receivable/Payable between participant and header.
    pub fn generate_cash_pool_sweep_jes(
        sweeps: &[CashPoolSweep],
        header_entity: &str,
    ) -> Vec<JournalEntry> {
        let mut jes = Vec::new();

        for sweep in sweeps {
            if sweep.amount == Decimal::ZERO {
                continue;
            }

            let mut je = JournalEntry::new_simple(
                format!("JE-TREAS-SWEEP-{}", sweep.id),
                header_entity.to_string(),
                sweep.date,
                format!("Cash pool sweep {} → {}", sweep.from_account_id, sweep.to_account_id),
            );

            je.add_line(JournalEntryLine {
                line_number: 1,
                gl_account: treasury_accounts::CASH_POOL_IC_RECEIVABLE.to_string(),
                debit_amount: sweep.amount,
                reference: Some(sweep.pool_id.clone()),
                ..Default::default()
            });
            je.add_line(JournalEntryLine {
                line_number: 2,
                gl_account: treasury_accounts::CASH_POOL_IC_PAYABLE.to_string(),
                credit_amount: sweep.amount,
                reference: Some(sweep.pool_id.clone()),
                ..Default::default()
            });
            jes.push(je);
        }

        jes
    }
}
```

```rust
fn compute_pre_tax_income(company_code: &str, entries: &[JournalEntry]) -> Decimal {
    let mut total_revenue = Decimal::ZERO;
    let mut total_expenses = Decimal::ZERO;

    for je in entries {
        if je.company_code() != company_code {
            continue;
        }
        for line in &je.lines {
            let acct = &line.gl_account;
            if acct.starts_with('4') {
                // Revenue: credit-normal
                total_revenue += line.credit_amount - line.debit_amount;
            } else if acct.starts_with('5') || acct.starts_with('6') || acct.starts_with('7') {
                // Expenses: debit-normal
                total_expenses += line.debit_amount - line.credit_amount;
            }
        }
    }

    total_revenue - total_expenses
}
```

```rust
//! Generates GL posting JEs from TaxLine records.
//!
//! For each TaxLine:
//! - Output tax (CustomerInvoice): DR AR/Receivable, CR VAT Payable
//! - Input tax (VendorInvoice, deductible): DR Input VAT, CR AP/Payable

use datasynth_core::accounts::{control_accounts, tax_accounts};
use datasynth_core::models::tax::{TaxLine, TaxableDocumentType};
use datasynth_core::models::{JournalEntry, JournalEntryLine};
use rust_decimal::Decimal;

pub struct TaxPostingGenerator;

impl TaxPostingGenerator {
    pub fn generate_tax_posting_jes(
        tax_lines: &[TaxLine],
        company_code: &str,
    ) -> Vec<JournalEntry> {
        let mut jes = Vec::new();

        for tl in tax_lines {
            if tl.tax_amount == Decimal::ZERO {
                continue;
            }

            let posting_date = chrono::Utc::now().date_naive(); // Use document date if available

            match tl.document_type {
                TaxableDocumentType::CustomerInvoice => {
                    // Output tax: DR AR, CR VAT Payable
                    let mut je = JournalEntry::new_simple(
                        format!("JE-TAX-OUT-{}", tl.id),
                        company_code.to_string(),
                        posting_date,
                        format!("Output VAT — {} ({})", tl.document_id, tl.tax_code_id),
                    );
                    je.add_line(JournalEntryLine {
                        line_number: 1,
                        gl_account: control_accounts::AR_CONTROL.to_string(),
                        debit_amount: tl.tax_amount,
                        reference: Some(tl.document_id.clone()),
                        ..Default::default()
                    });
                    je.add_line(JournalEntryLine {
                        line_number: 2,
                        gl_account: tax_accounts::VAT_PAYABLE.to_string(),
                        credit_amount: tl.tax_amount,
                        reference: Some(tl.document_id.clone()),
                        ..Default::default()
                    });
                    jes.push(je);
                }
                TaxableDocumentType::VendorInvoice if tl.is_deductible => {
                    // Input tax: DR Input VAT, CR AP
                    let mut je = JournalEntry::new_simple(
                        format!("JE-TAX-IN-{}", tl.id),
                        company_code.to_string(),
                        posting_date,
                        format!("Input VAT — {} ({})", tl.document_id, tl.tax_code_id),
                    );
                    je.add_line(JournalEntryLine {
                        line_number: 1,
                        gl_account: tax_accounts::INPUT_VAT.to_string(),
                        debit_amount: tl.tax_amount,
                        reference: Some(tl.document_id.clone()),
                        ..Default::default()
                    });
                    je.add_line(JournalEntryLine {
                        line_number: 2,
                        gl_account: control_accounts::AP_CONTROL.to_string(),
                        credit_amount: tl.tax_amount,
                        reference: Some(tl.document_id.clone()),
                        ..Default::default()
                    });
                    jes.push(je);
                }
                _ => {} // Other document types — no GL posting needed
            }
        }

        jes
    }
}
```

Note: The `posting_date` should ideally come from the source document. Since `TaxLine` doesn't carry a date, use `chrono::Utc::now().date_naive()` as a fallback, or accept a date parameter. Read the actual `TaxLine` struct to check if there's a date field — if so, use it.

Add a new method to `PayrollGenerator` in `payroll_generator.rs`:

```rust
/// Generate payroll with employee change history applied.
///
/// For each employee, checks if there are salary adjustments or terminations
/// effective within or before the pay period, and adjusts the base salary accordingly.
pub fn generate_with_changes(
    &mut self,
    company_code: &str,
    employees: &[(String, Decimal, Option<String>, Option<String>)],
    period_start: NaiveDate,
    period_end: NaiveDate,
    currency: &str,
    changes: &[EmployeeChangeEvent],
) -> (PayrollRun, Vec<PayrollLineItem>) {
    // Build adjusted employee list
    let adjusted: Vec<_> = employees.iter().map(|(id, salary, cc, dept)| {
        let adjusted_salary = Self::apply_changes(id, *salary, period_start, period_end, changes);
        (id.clone(), adjusted_salary, cc.clone(), dept.clone())
    }).collect();

    self.generate(company_code, &adjusted, period_start, period_end, currency)
}

fn apply_changes(
    employee_id: &str,
    base_annual_salary: Decimal,
    period_start: NaiveDate,
    period_end: NaiveDate,
    changes: &[EmployeeChangeEvent],
) -> Decimal {
    // Find the most relevant salary adjustment for this employee
    let salary_changes: Vec<_> = changes.iter()
        .filter(|c| c.employee_id == employee_id
            && matches!(c.event_type, EmployeeEventType::SalaryAdjustment)
            && c.effective_date <= period_end)
        .collect();

    if salary_changes.is_empty() {
        return base_annual_salary;
    }

    // Get the latest change effective on or before period_end
    let latest = salary_changes.iter()
        .max_by_key(|c| c.effective_date)
        .unwrap();

    let new_salary: Decimal = latest.new_value.as_deref()
        .and_then(|v| v.parse().ok())
        .unwrap_or(base_annual_salary);

    if latest.effective_date <= period_start {
        // Change was before period start — use new salary for full period
        new_salary
    } else {
        // Mid-period change — prorate
        let total_days = (period_end - period_start).num_days() as f64;
        let days_at_old = (latest.effective_date - period_start).num_days() as f64;
        let days_at_new = total_days - days_at_old;

        let old_fraction = Decimal::from_f64_retain(days_at_old / total_days).unwrap_or(Decimal::ZERO);
        let new_fraction = Decimal::from_f64_retain(days_at_new / total_days).unwrap_or(Decimal::ZERO);

        (base_annual_salary * old_fraction + new_salary * new_fraction).round_dp(2)
    }
}
```
