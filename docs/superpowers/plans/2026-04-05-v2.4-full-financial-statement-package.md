- [ ] **Step 1: Write tests**

Create `crates/datasynth-generators/tests/dividend_test.rs`:

```rust
use datasynth_generators::period_close::DividendGenerator;
use datasynth_core::accounts::{dividend_accounts, equity_accounts, cash_accounts};
use rust_decimal_macros::dec;
use rust_decimal::Decimal;
use chrono::NaiveDate;

#[test]
fn test_dividend_declaration_je() {
    let mut gen = DividendGenerator::new(42);
    let result = gen.generate(
        "C001",
        NaiveDate::from_ymd_opt(2025, 3, 31).unwrap(), // declaration date
        dec!(2.50),    // per share
        dec!(100_000), // total amount
        "USD",
    );

    assert_eq!(result.declarations.len(), 1);
    let decl = &result.declarations[0];
    assert_eq!(decl.total_amount, dec!(100_000));
    assert!(decl.declaration_date <= decl.record_date);
    assert!(decl.record_date <= decl.payment_date);

    // Should have 2 JEs: declaration and payment
    assert_eq!(result.journal_entries.len(), 2);
    for je in &result.journal_entries {
        assert!(je.is_balanced());
    }

    // Declaration JE: DR Retained Earnings, CR Dividends Payable
    let decl_je = &result.journal_entries[0];
    let has_re_debit = decl_je.lines.iter()
        .any(|l| l.gl_account == equity_accounts::RETAINED_EARNINGS && l.debit_amount > Decimal::ZERO);
    let has_dp_credit = decl_je.lines.iter()
        .any(|l| l.gl_account == dividend_accounts::DIVIDENDS_PAYABLE && l.credit_amount > Decimal::ZERO);
    assert!(has_re_debit, "Should debit Retained Earnings");
    assert!(has_dp_credit, "Should credit Dividends Payable");

    // Payment JE: DR Dividends Payable, CR Cash
    let pay_je = &result.journal_entries[1];
    let has_dp_debit = pay_je.lines.iter()
        .any(|l| l.gl_account == dividend_accounts::DIVIDENDS_PAYABLE && l.debit_amount > Decimal::ZERO);
    let has_cash_credit = pay_je.lines.iter()
        .any(|l| l.gl_account == cash_accounts::OPERATING_CASH && l.credit_amount > Decimal::ZERO);
    assert!(has_dp_debit, "Should debit Dividends Payable");
    assert!(has_cash_credit, "Should credit Cash");
}
```

- Create: `crates/datasynth-eval/tests/financial_package_coherence_test.rs`

- [ ] **Step 1: Implement four capstone validators**

**1. CashFlowReconciliationEvaluator**
- Input: opening_cash, net_cash_flows (sum of operating + investing + financing), closing_cash_gl
- Reconciled if: opening_cash + net_cash_flows ≈ closing_cash_gl (zero tolerance)

**2. EquityRollforwardEvaluator**
- Input: opening_equity, net_income, oci_movements, dividends_declared, stock_comp, closing_equity
- Reconciled if: opening + NI + OCI - dividends + stock_comp ≈ closing

**3. SegmentReconciliationEvaluator**
- Input: sum_segment_revenue, ic_eliminations, consolidated_revenue
- Reconciled if: sum_segments - eliminations ≈ consolidated

**4. TrialBalanceMasterProofEvaluator**
- Input: sum_opening_balances, sum_all_je_debits, sum_all_je_credits, closing_tb_total_debits, closing_tb_total_credits
- Reconciled if: opening + debits - credits ≈ closing (per side)
- This is the master proof — if it passes, every generator's GL output is accounted for
